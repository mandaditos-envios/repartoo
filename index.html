// Escuchador botón Añadir Serie document.getElementById('btn-add-log').addEventListener('click', function() { const muscle = document.getElementById('muscle').value; const exercise = document.getElementById('exercise').value.trim() || 'Ejercicio'; const reps = document.getElementById('reps').value; const weight = document.getElementById('weight').value || 0; if (!reps) { alert('Ingresa la cantidad de repeticiones.'); return; } logs.push({ muscle, exercise, reps, weight }); document.getElementById('reps').value = ''; document.getElementById('weight').value = ''; renderLogs(); }); // Escuchador botón Consultar IA document.getElementById('btn-consult-ai').addEventListener('click', async function() { const apiKey = localStorage.getItem('coach_gemini_key') || document.getElementById('api-key').value.trim(); if (!apiKey) { alert('Ingresa y guarda tu API Key de Gemini primero.'); return; } const aiBtn = document.getElementById('btn-consult-ai'); const resBox = document.getElementById('ai-response'); aiBtn.disabled = true; aiBtn.innerText = 'Conectando con Gemini...'; resBox.style.display = 'block'; resBox.innerText = 'Analizando tus series registradas...'; const prompt = "Actúa como un entrenador experto. Analiza este entrenamiento y dame 2 consejos breves de sobrecarga progresiva:\n" + JSON.stringify(logs, null, 2); try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); const data = await response.json(); if (data.candidates && data.candidates[0]?.content?.parts[0]?.text) { resBox.innerText = data.candidates[0].content.parts[0].text; } else if (data.error) { resBox.innerText = '❌ Error de la API:\n' + (data.error.message || JSON.stringify(data.error)); } else { resBox.innerText = '❌ No se obtuvo respuesta del modelo.'; } } catch (err) { resBox.innerText = '❌ Error de conexión:\n' + err.message; } finally { aiBtn.disabled = false; aiBtn.innerText = '✨ Analizar mi Entrenamiento con IA'; } }); }); function renderLogs() { const container = document.getElementById('logs-container'); if (logs.length === 0) { container.innerHTML = 'No hay series registradas aún.'; return; } container.innerHTML = logs.map((l, idx) => `
${l.muscle} (${l.exercise}) ${l.reps} reps @ ${l.weight} kg
`).join(''); }
