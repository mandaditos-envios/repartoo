async function askAI() { if (!apiKey) return alert('Guarda tu API Key primero'); const box = document.getElementById('ai-res'); box.style.display = 'block'; box.innerText = '🤖 Analizando con Gemini...'; const prompt = `Analiza mi entrenamiento y da consejos breves de sobrecarga progresiva:\n${JSON.stringify(history)}`; // Lista de endpoints a probar (v1 y v1beta con modelos compatibles) const endpoints = [ `https://generativelanguage.googleapis.com/v1/models/gemini-1.5-flash:generateContent`, `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash-latest:generateContent`, `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-pro:generateContent` ]; let lastError = ''; for (const url of endpoints) { try { const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-goog-api-key': apiKey // Soporte nativo para las nuevas llaves AQ. }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); const data = await res.json(); if (data.candidates?.[0]?.content?.parts?.[0]?.text) { box.innerText = data.candidates[0].content.parts[0].text; return; // Éxito, salir de la función } else if (data.error) { lastError = data.error.message || JSON.stringify(data.error); } } catch(e) { lastError = e.message; } } box.innerText = '❌ Error al conectar:\n' + lastError; }
